### سوال ۱
### الف) 

```sql
SELECT s.name
FROM Students s
JOIN Enrollments e ON s.id = e.student_id
GROUP BY s.id, s.name
HAVING COUNT(e.course_id) > 2;
```

### ب) 

```sql
SELECT c.course_name, COUNT(DISTINCT e.student_id) AS student_count
FROM Courses c
JOIN Enrollments e ON c.course_id = e.course_id
JOIN Grades g ON e.student_id = g.student_id AND e.course_id = g.course_id
GROUP BY c.course_id, c.course_name
HAVING AVG(g.grade) > 10;
```

### ج)

```sql
SELECT s.name
FROM Students s
WHERE s.id NOT IN (
    SELECT e.student_id
    FROM Enrollments e
    JOIN Courses c ON e.course_id = c.course_id
    WHERE c.course_name IN ('DB', 'OS')
);
```

## سوال 2:

### الف) 

```sql
SELECT d.department_name, AVG(s.salary) AS avg_salary
FROM Departments d
JOIN Employees e ON d.department_id = e.department_id
JOIN Salaries s ON e.employee_id = s.employee_id
GROUP BY d.department_id, d.department_name;
```

### ب)

```sql
SELECT d.department_name
FROM Departments d
WHERE d.department_id NOT IN (
    SELECT DISTINCT e.department_id
    FROM Employees e
    WHERE e.department_id IS NOT NULL
);
```

### ج) 

```sql
SELECT d.department_name, COUNT(e.employee_id) AS employee_count
FROM Departments d
JOIN Employees e ON d.department_id = e.department_id
GROUP BY d.department_id, d.department_name
HAVING COUNT(e.employee_id) > 10;
```

### د) 

```sql
SELECT e.employee_id, e.name, e.job_title
FROM Employees e
JOIN Salaries s ON e.employee_id = s.employee_id
WHERE s.salary = (
    SELECT salary
    FROM Salaries
    WHERE employee_id = '110'
)
AND e.employee_id <> '110';
```

## سوال 3: 

### الف) 

```sql
SELECT COUNT(*) AS order_count
FROM Orders o
JOIN Products p ON o.product_id = p.product_id
WHERE p.product_name = 'Laptop';
```

### ب) 

```sql
SELECT DISTINCT c.customer_name
FROM Customers c
JOIN Orders o ON c.customer_id = o.customer_id
JOIN Products p ON o.product_id = p.product_id
JOIN Categories cat ON p.category_id = cat.category_id
WHERE cat.category_name = 'Electronics';
```

### ج) 

```sql
SELECT p.product_name, cat.category_name, COUNT(o.order_id) AS order_count
FROM Products p
JOIN Categories cat ON p.category_id = cat.category_id
JOIN Orders o ON p.product_id = o.product_id
GROUP BY p.product_id, p.product_name, cat.category_name
HAVING COUNT(o.order_id) > 50;
```

### د) 

```sql
SELECT c.customer_id, c.customer_name, COUNT(o.order_id) AS order_count
FROM Customers c
JOIN Orders o ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.customer_name
HAVING COUNT(o.order_id) > 3;
```

### ه) 


```sql
SELECT c.customer_name
FROM Customers c
JOIN Orders o ON c.customer_id = o.customer_id
WHERE c.customer_id IN (
    SELECT DISTINCT o2.customer_id
    FROM Orders o2
    JOIN Products p ON o2.product_id = p.product_id
    JOIN Categories cat ON p.category_id = cat.category_id
    WHERE cat.category_name = 'Electronics'
)
GROUP BY c.customer_id, c.customer_name
HAVING COUNT(o.order_id) > 2;
```
